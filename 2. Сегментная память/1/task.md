# Преобразование адреса в четырёхсегментной памяти

В компьютере используется четырёхсегментная модель организации памяти. Виртуальные и физические адреса имеют размер 32 бита; память адресуется побайтно. Используются все четыре сегмента, и все они находятся в физической памяти.

Напишите программу, преобразующую виртуальный сегментный адрес в физический. Два старших бита виртуального адреса задают номер сегмента, остальные 30 бит — смещение от его начала. Таблица сегментов должна содержать физический адрес начала `base` и размер `size` каждого сегмента в байтах. Допустимые смещения: `0 <= offset < size`.

Если `offset >= size`, процесс должен послать **самому себе `SIGUSR1`**. После такой ошибки нельзя возвращать вычисленный физический адрес, даже если сигнал проигнорирован, заблокирован или установленный обработчик вернулся.

Опишите все необходимые структуры данных. Предполагается, что таблица корректна: размер сегмента не превышает `2^30`, а весь сегмент помещается в 32-битное физическое адресное пространство. Проверять корректность самой таблицы не требуется.

Исходное условие не задаёт формат ввода. **Принятое соглашение для законченной программы:** таблица задана в коде, единственный аргумент командной строки — виртуальный адрес в шестнадцатеричной записи. Успешный результат печатается восемью шестнадцатеричными цифрами без префикса `0x`. Используется POSIX-среда с `SIGUSR1`. Если после отправки сигнала управление продолжается, программа немедленно завершается с `EXIT_FAILURE`; ложный физический адрес не возвращается. Это соглашение уточняет отсутствующий в исходнике сценарий возвращающегося обработчика.

Источник: «Задачи_вирт_память.docx», задача о четырёх сегментах; фотографии 21-11-53 и 21-11-57. Размер адреса 32 бита зафиксирован для этого варианта вместо зависимости от `sizeof(int)` конкретного компьютера.

---

# решение

Номер сегмента получается сдвигом вправо на 30 бит. Маска `0x3fffffff` оставляет только смещение. После проверки границы физический адрес равен `base + offset`.

В отличие от наброска с доски, недостаточно просто вызвать `kill`, а затем продолжить вычисление: обработчик сигнала может вернуть управление. Отдельная функция `_Noreturn` отправляет сигнал и, если управление сохранилось, вызывает `_Exit`. Сигнал по-прежнему посылается именно текущему процессу через `kill(getpid(), SIGUSR1)`.

Код — полная программа на C11 для POSIX-системы. Тип `uint32_t` фиксирует ширину адресов независимо от размера `int` на машине, где запускается решение.

```c
#define _POSIX_C_SOURCE 200809L
#include <errno.h>
#include <inttypes.h>
#include <signal.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

typedef struct {
    uint32_t base;  /* Физический адрес первого байта. */
    uint32_t size;  /* Размер сегмента в байтах. */
} segment;

static _Noreturn void invalid_segment(void)
{
    (void)kill(getpid(), SIGUSR1);
    _Exit(EXIT_FAILURE);
}

uint32_t VirtIntoPhys(const segment table[4], uint32_t virtual_address)
{
    uint32_t number = virtual_address >> 30;
    uint32_t offset = virtual_address & UINT32_C(0x3fffffff);

    if (offset >= table[number].size) {
        invalid_segment();
    }
    return table[number].base + offset;
}

int main(int argc, char **argv)
{
    const segment table[4] = {
        {UINT32_C(0x1000), UINT32_C(0x100)},
        {UINT32_C(0x2000), UINT32_C(0x080)},
        {UINT32_C(0x3000), UINT32_C(0x020)},
        {UINT32_C(0x4000), UINT32_C(0x010)}
    };

    if (argc != 2) {
        fprintf(stderr, "Usage: %s virtual_address_hex\n", argv[0]);
        return EXIT_FAILURE;
    }

    char *end;
    errno = 0;
    uintmax_t address = strtoumax(argv[1], &end, 16);
    if (argv[1][0] == '-' || end == argv[1] || *end != '\0' ||
        errno != 0 || address > UINT32_MAX) {
        fputs("Invalid 32-bit hexadecimal address\n", stderr);
        return EXIT_FAILURE;
    }

    uint32_t physical_address = VirtIntoPhys(table, (uint32_t)address);
    if (printf("%08" PRIX32 "\n", physical_address) < 0) {
        return EXIT_FAILURE;
    }
    return EXIT_SUCCESS;
}
```
